## 一、什么是IO多路复用

### 传统简单Socket通信的问题

在传统的Socket服务端编程中，通常采用以下模式：

```c
while(1) {
    client_fd = accept(server_fd, ...);
    // 处理客户端请求
    read(client_fd, buffer, size);
    // 处理数据
    write(client_fd, response, size);
    close(client_fd);
}
```

这种模式存在以下问题：

- 服务器在accept()和read()时会阻塞，无法在一个线程里同时处理多个客户端
- 为了支持多客户端，需要创建多个进程或线程，消耗大量系统资源

### IO多路复用的概念

IO多路复用是一种IO模型，基于事件循环，允许单个线程同时监控多个文件描述符（包括socket），当其中任何一个文件描述符发生对应事件的时候，系统会通知应用程序。这样可以在单线程中高效地处理多个并发连接。

**主要优势：**

- 单线程处理多个连接，减少上下文切换开销
- 避免了多线程/多进程的复杂性和资源消耗
- 提高了服务器的并发处理能力

### IO多路复用的读写事件

#### 什么是读事件

读事件表示文件描述符的接收缓冲区上有数据可供读取，或者连接状态发生了可读的变化。

##### 读事件的具体情况：

**套接字有数据到达**

- TCP连接接收到新数据
- UDP套接字接收到数据包

** 连接建立**

- 服务器套接字有新的客户端连接请求（可以调用accept()）

#### 什么是写事件

写事件表示文件描述符可以进行写操作而不会阻塞。

##### 写事件的具体情况：

**套接字发送缓冲区有空间 **

- TCP套接字的发送缓冲区不满，可以写入数据
- UDP套接字可以发送数据包

## 二、UNIX通用的IO多路复用技术

### 1. select

#### 概念及特点

`select`是最早的IO多路复用机制，它允许程序监控多个文件描述符，等待其中任何一个变为"准备好"的状态。

**特点：**

- 数据结构基于位图存储，有文件描述符数量限制（通常为1024）

**相关宏定义：**

- `FD_ZERO(fd_set *set)` - 清空文件描述符集合
- `FD_SET(int fd, fd_set *set)` - 将fd添加到集合中
- `FD_CLR(int fd, fd_set *set)` - 从集合中移除fd
- `FD_ISSET(int fd, fd_set *set)` - 检查fd是否在集合中

#### 相关结构体

```c
typedef struct {
    long fds_bits[FD_SETSIZE/BITSPERLONG];
} fd_set;
```

其中：

- `__FD_SETSIZE` 通常为 1024，表示最大文件描述符数量
- `__NFDBITS` 通常为 64（在64位系统上），表示每个 long int 的位数
- 因此 `__fds_bits` 数组包含 16 个 long int 元素（1024/64=16）

##### 位图存储原理-类似单片机编程里的寄存器操作

**位映射机制**：fd_set 使用位图来表示文件描述符集合，每个文件描述符对应位图中的一个位。如果某个文件描述符在集合中，对应的位被设置为1，否则为0。

**存储布局**：

- 文件描述符 fd 对应的位位于 `__fds_bits[fd / __NFDBITS]` 这个 long int 中
- 在该 long int 内的具体位置是 `fd % __NFDBITS`
- 例如，fd=65 对应 `__fds_bits[1]` 的第1位（65/64=1, 65%64=1）

##### 位操作实现

**设置位**：`__fds_bits[fd/__NFDBITS] |= (1UL << (fd % __NFDBITS))`
**清除位**：`__fds_bits[fd/__NFDBITS] &= ~(1UL << (fd % __NFDBITS))`
**检查位**：`(__fds_bits[fd/__NFDBITS] & (1UL << (fd % __NFDBITS))) != 0`

#### 通用编写流程

- 创建服务端监听的socket
- 创建监听位图，将监听socket添加到位图上，并且设置成最大fd
- while死循环
     1. 定义一个新的位图接收select的结果
     2. 调用select
           - 如果select返回值<0,报错
           - 如果select返回值=0,下一轮循环
           - 如果select返回值>0
                - 从0遍历到maxfd
                - 依次判断是否在位图里
                     - 如果不在，下一轮循环
                     - 如果在,,判断fd
                          - 如果等于监听的socket，接受客户端连接，添加到监听位图，更新maxfd
                          - 处理已连接客户端的事件

#### 代码示例

```c
#include <arpa/inet.h>
#include <netinet/in.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/select.h>
#include <sys/socket.h>
#include <unistd.h>

// 连接结构体
struct connection {
  int sock;
  struct sockaddr_in addr;
};

// 服务端抽象结构体
struct server {
  int (*make_listener)(const char *, int);
  struct connection (*accept_connection)(int);
  int (*echo)(int, void (*)(int, void *), void *);
  void (*poll)(int);
};

int make_listener(const char *ip, int port) {
  // 1.创建socket
  int listen_sock = socket(AF_INET, SOCK_STREAM, 0);
  if (listen_sock == -1) {
    perror("socket");
    return -1;
  }
  // 2.构造地址结构体
  struct sockaddr_in addr;
  addr.sin_family = AF_INET;
  addr.sin_addr.s_addr = inet_addr(ip);
  addr.sin_port = htons(port);
  // 3.绑定
  if (bind(listen_sock, (struct sockaddr *)&addr, sizeof(addr)) == -1) {
    perror("bind");
    close(listen_sock);
    return -1;
  }
  // 4.监听
  if (listen(listen_sock, 5) == -1) {
    perror("listen");
    close(listen_sock);
    return -1;
  }
  return listen_sock;
}

struct connection accept_connection(int listen_sock) {
  struct connection conn;
  conn.sock = -1;
  // 第二阶段 连接建立
  socklen_t addr_len = sizeof(conn.addr);
  int sock = accept(listen_sock, (struct sockaddr *)&conn.addr, &addr_len);
  if (sock == -1) {
    perror("accept");
    return conn;
  }
  conn.sock = sock;
  printf("Client connected: %s:%d\n", inet_ntoa(conn.addr.sin_addr),
         ntohs(conn.addr.sin_port));
  return conn;
}

int echo(int sock, void (*__callback)(int, void *args), void *args) {
  char buffer[1024];
  ssize_t bytes_received = recv(sock, buffer, sizeof(buffer), 0);
  if (bytes_received == -1) {
    perror("recv");
    close(sock);
    return -1;
  }

  if (bytes_received == 0) {
    if (__callback != NULL) {
      __callback(sock, args);
    } else {
      close(sock);
    }
    return 0;
  }

  buffer[bytes_received] = '\0'; // 确保字符串以null结尾
  printf("Received from client: %s\n", buffer);
  int bytes_sended = send(sock, buffer, bytes_received, 0);
  if (bytes_sended == -1) {
    perror("send");
    close(sock);
    return -1;
  }

  if (bytes_sended == 0) {
    if (__callback != NULL) {
      __callback(sock, args);
    } else {
      close(sock);
    }
    return 0;
  }
  return 1;
}

//回调函数参数结构体
struct args {
  fd_set *read_fds;
  int *maxfd;
};

void callback(int sock, void *args) {
  struct args *cbargs = (struct args *)args;
  close(sock);
  // 将sock从位图里删除
  FD_CLR(sock, cbargs->read_fds);
  // 如果删除的sock正好是maxfd,更新maxfd
  // 方法：从maxfd向下递减尝试
  if (sock == *cbargs->maxfd) {
    for (int i = *cbargs->maxfd - 1; i > 0; i--) {
      if (FD_ISSET(i, cbargs->read_fds)) {
        *cbargs->maxfd = i;
        break;
      }
    }
  }
}

void poll(int listen_sock) {
  // 定义一个fd_set 位图
  fd_set read_fds;
  // 清空位图
  FD_ZERO(&read_fds);
  // 将监听socket加入位图
  FD_SET(listen_sock, &read_fds);

  // 定义最大的fd，也是fd_set位图的最大值，初始设置成监听socket
  int maxfd = listen_sock;
  while (1) {
    // 定义一个结果位图
    fd_set result_fds;
    // 将监听位图赋值给结果位图
    result_fds = read_fds;
    // 调用select，阻塞等待有事件的fd返回
    int flag = select(maxfd + 1, &result_fds, NULL, NULL, NULL);
    if (flag < 0) {
      perror("select");
      close(listen_sock);
      break;
    }
    // 超时
    if (flag == 0)
      continue;
    // 处理有事件的socket
    for (int fd = 0; fd <= maxfd; fd++) {
      // 判断socket是否在结果位图里,如果不在直接跳入下一轮循环
      if (!FD_ISSET(fd, &result_fds))
        continue;

      // 如果fd是监听的fd，那说明有客户端连接了
      // 接受连接，并且把客户端的socket加入监听的位图中
      if (fd == listen_sock) {
        struct connection conn = accept_connection(listen_sock);
        // 添加到监听位图里
        FD_SET(conn.sock, &read_fds);
        // 如果client_sock比最大的fd还大，更新maxfd
        if (conn.sock > maxfd)
          maxfd = conn.sock;
        continue;
      } else {
        // 否则处理客户端连接
          struct args cbargs = {.maxfd = &maxfd, .read_fds = &read_fds};
          int res;
          res = echo(fd, callback, &cbargs);
          if (res <= 0) {
            break;
        }
      }
    }
  }
}

// 创建server结构体
struct server make_server() {
  struct server server = {
      .make_listener = make_listener,
      .accept_connection = accept_connection,
      .echo = echo,
      .poll = poll,
  };
  return server;
};

int main() {
  struct server server = make_server();
  int listen_sock = server.make_listener("127.0.0.1", 8080);
  if (listen_sock == -1) {
    return -1;
  }

  server.poll(listen_sock);

  close(listen_sock);
  return 0;
}
```

##### 对代码的解释

###### 1. 为什么要设置一个maxfd，以及为什么maxfd会作为位图遍历的边界

**因为select的工作机制，select函数需要maxfd**

```c
int select(int nfds, fd_set *readfds, 
fd_set *writefds, 
           fd_set *exceptfds, struct 
           timeval *timeout);
```

- nfds 参数：需要检查的文件描述符数量，通常设置为 maxfd + 1
- 内核只检查 0 到 nfds-1 范围内的文件描述符

###### 2. 为什么要设计两个位图read_fds和result_fds

**核心原因：select会破坏性地修改传入的fd_set**
select函数会修改传入的fd_set，将其从"监控集合"转换为"就绪集合" 。

###### 3.移除socket的时候重新计算maxfd的逻辑

**本质是从maxfd开始逐个递减尝试，如果在递减的过程中有值在位图里，那就设置成maxfd**

### 2. poll

#### 概念及特点

`poll`是select的改进版本，使用结构体数组来管理文件描述符，克服了select的一些限制。

**特点：**

- 基于结构体数组，没有文件描述符数量限制（除了系统限制）

#### 相关结构体

```c
struct pollfd {
    int fd;         // 文件描述符
    short events;   // 请求的事件
    short revents;  // 返回的事件
};
```

**事件类型：**

- `POLLIN` - 有数据可读
- `POLLOUT` - 可以写数据
- `POLLERR` - 发生错误
- `POLLHUP` - 连接挂起
- `POLLNVAL` - 文件描述符无效

#### 通用编写流程

- 创建服务端监听的socket
- 创建pollfd结构体数组，将监听socket添加到数组中，并且设置成最大fd
- while死循环
     1. 调用poll
           - 如果poll返回值<0,报错退出循环
           - 如果poll返回值=0，说明超时了，跳到下一轮循环
           - 如果poll返回值>0，从0遍历到maxfd
                - 依次判断文件是否在pollfd结构体数组里，并且判断fd返回事件是否有读事件
                     - 如果fd不在pollfd结构体数组里或者fd没有返回读事件，跳到下一轮循环
                     - 否则，重置fd返回事件,判断fd
                          - 如果等于监听的socket，接受客户端连接，添加到pollfd结构体数组里，更新maxfd
                          - 否则处理已连接客户端的读事件

#### 代码示例

```c

#include <netinet/in.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/poll.h>
#include <sys/socket.h>
#include <unistd.h>
// 连接结构体
struct connection {
  int sock;
  struct sockaddr_in addr;
};

// 服务端抽象结构体
struct server {
  int (*make_listener)(const char *, int);
  struct connection (*accept_connection)(int);
  int (*echo)(int, void (*)(int, void *), void *);
  void (*poll)(int);
};

int make_listener(const char *ip, int port) {

  int listen_sock = socket(AF_INET, SOCK_STREAM, 0);
  if (listen_sock == -1) {

struct connection accept_connection(int listen_sock) {

  conn.sock = -1;

  socklen_t addr_len = sizeof(conn.addr);
  int sock = accept(listen_sock, (struct sockaddr *)&conn.addr, &addr_len);
  if (sock == -1) {

  conn.sock = sock;
  printf("Client connected: %s:%d\n", inet_ntoa(conn.addr.sin_addr),

  return conn;
}

int echo(int sock, void (*__callback)(int, void *args), void *args) {
  char buffer[1024];
  ssize_t bytes_received = recv(sock, buffer, sizeof(buffer), 0);
  if (bytes_received == -1) {
    perror("recv");
    close(sock);
    return -1;
  }

  if (bytes_received == 0) {
    if (__callback != NULL) {
      __callback(sock, args);
    } else {
      close(sock);
    }
    return 0;
  }

  buffer[bytes_received] = '\0'; // 确保字符串以null结尾
  printf("Received from client: %s\n", buffer);
  int bytes_sended = send(sock, buffer, bytes_received, 0);
  if (bytes_sended == -1) {
    perror("send");
    close(sock);
    return -1;
  }

  if (bytes_sended == 0) {
    if (__callback != NULL) {
      __callback(sock, args);
    } else {
      close(sock);
    }
    return 0;
  }
  return 1;
}

// 回调函数参数结构体
struct args {
  struct pollfd *fds;
  int *maxfd;
};

void callback(int sock, void *args) {
  struct args *cbargs = (struct args *)args;
  close(sock);
  cbargs->fds[sock].fd = -1;
  if (sock == *cbargs->maxfd) {
    for (int i = *cbargs->maxfd - 1; i > 0; i--) {
      if (cbargs->fds[i].fd == i) {
        *cbargs->maxfd = i;
        break;
      }
    }
  }
}

void poll1(int listen_sock) {
  // 创建一个pollfd结构体数组
  struct pollfd fds[1024];
  // 初始化每一个结构体的fd都是-1
  for (int i = 0; i < 1024; i++) {
    fds[i].fd = -1;
  }
  // 将监听的socket加入fds数组
  fds[listen_sock].fd = listen_sock;
  fds[listen_sock].events = POLLIN;
  fds[listen_sock].revents = 0;
  // 定义最大的fd，也是fds数组里监听文件描述符的最大值，初始设置成监听socket
  int maxfd = listen_sock;
  while (1) {
    // 调用poll，阻塞等待有事件发生，poll返回
    int flag = poll(fds, maxfd + 1, 1000);
    if (flag < 0) {
      perror("poll");
      close(listen_sock);
      break;
    }
    if (flag == 0)
      continue;
    // 处理有事件的socket
    for (int fd = 0; fd <= maxfd; fd++) {
      // 判断socket是否在fds数组里和是否有读事件,直接跳入下一轮循环
      if (fds[fd].fd < 0 || fds[fd].revents != POLLIN)
        continue;
      // 重置fd的返回事件
      fds[fd].revents = 0;
      // 如果fd是监听的fd，那说明有客户端连接了
      // 接受连接，并且把客户端的socket加入fds数组中
      if (fd == listen_sock) {
        struct connection conn = accept_connection(listen_sock);
        if (conn.sock == -1) {
          continue;
        }
        // 添加到pollfd数组
        fds[conn.sock].fd = conn.sock;
        fds[conn.sock].events = POLLIN;
        fds[conn.sock].revents = 0;

        // 如果client_sock比最大的fd还大，更新maxfd
        if (conn.sock > maxfd)
          maxfd = conn.sock;
        continue;
      } else {
          struct args cbargs = {.fds = fds, .maxfd = &maxfd};
          int res;
          res = echo(fd, callback, &cbargs);
          if (res <= 0) {
            break;
          }
      }
    }
  }
}

// 创建server结构体
struct server make_server() {
  struct server server = {
      .make_listener = make_listener,
      .accept_connection = accept_connection,
      .echo = echo,
      .poll = poll1,
  };
  return server;
};

int main() {
  struct server server = make_server();
  int listen_sock = server.make_listener("127.0.0.1", 8080);
  if (listen_sock == -1) {
    return -1;
  }

  server.poll(listen_sock);

  close(listen_sock);
  return 0;
}
```

##### 对代码的解释

###### 1. 为什么pollfd结构体数组不需要两个

**因为poll不会改变fd，只是更改revent**

###### 2. 关于pollfd结构体数组相关设计

- **初始化：fd值全是-1**
- **添加：采用下标就是fd值的思路**

好处:
1.修改poll数组不需要移动其他元素进行重排序
2.删除元素只需要直接将poll数组对应下标的值设置成-1即可

### 3. select与poll的对比

| 特性               | select             | poll                 |
| ------------------ | ------------------ | -------------------- |
| **文件描述符限制** | 有限制（通常1024） | 无限制（除系统限制） |
| **数据结构**       | fd_set位图         | pollfd结构体数组     |
